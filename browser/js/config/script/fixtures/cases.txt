=== tour-crop
--- script
# A tour of the single-source form: open an image, edit it, save it.
# Comments start with '#'. Directives start with '@' and ignore case.

@SOURCE ./somepath/someimage.png:
    # Crop by named edges. A leading '-' measures from the far side.
    @CROP x1=10% x2=-10% y1=2cm y2=-1in

    # Or by insets: one value for all four sides, two for x and y,
    # four for x1 y1 x2 y2.
    @crop 10% 20cm

    # bw | sepia | invert | contour | none, or any colour for a duotone.
    @filter bw

    # The default unit for bare numbers. 23 means 23px until this changes.
    @use px

    # A line takes two or more points. A unit may sit on a component or
    # on the whole pair.
    @line (23, 14) (14px, 50%) (56, 90)cm, (-5, -89)%

    # Style settings apply to every shape after them, in any order.
    @use line #cccccc dashed, fill aqua, point red 2px

    # A rectangle from two opposite corners.
    @RECT (23, 14) (56, 90)

    # Bare @save writes next to the source as someimage-stencil.png.
    @save
--- dump
block 0 file "./somepath/someimage.png"
  op open "./somepath/someimage.png" 1
  op crop edit=1 "" [10% -10% 2cm -1in] 0
  op crop edit=2 "" [10% -10% 20cm -20cm] 0
  op filter edit=3 "bw" ""
  op line edit=4 "#FFFF00" "solid" "transparent" "" [23px 14px 14px 50% 56cm 90cm -5% -89%] 2 4 0
  op rect edit=5 "#cccccc" "dashed" "aqua" "red" [23px 14px 56px 90px] 2 2 1
  op save ""
=== tour-multi-source
--- script
# Several sources in one script. Each @source block is independent:
# its own edits, its own undo history, its own saves.

@SOURCE https://example.com/one.png:
    @rect (23, 14) (56, 90)
    @use stencil line-style:
    # On the editors a bare @save saves the open project; a name renames it.
    @save

@SOURCE https://example.com/two.png:
    @rect (23, 14) (56, 90)
    @save project2

# A video source: @frame picks the frame, and starts a fresh set of edits.
@source ./clip.mp4:
    @frame 8
    @rect (10, 10) (90, 60)
    @save
    @frame 90
    @filter sepia
    @save

# A whole directory, or a glob: every image gets the same treatment.
@source ./shots/:
    @filter bw
    @rect (23, 14) (56, 90)
    @save

@source ./shots/a*.png:
    @filter sepia
    @save ./out/

@stencil line-style:
    @use line #cccccc dashed, fill aqua, point red 2px
--- dump
block 0 url "https://example.com/one.png"
  op open "https://example.com/one.png" 2
  op rect edit=1 "#FFFF00" "solid" "transparent" "" [23px 14px 56px 90px] 2 4 1
  op save ""
block 1 url "https://example.com/two.png"
  op open "https://example.com/two.png" 2
  op rect edit=1 "#FFFF00" "solid" "transparent" "" [23px 14px 56px 90px] 2 4 1
  op save "project2"
block 2 file "./clip.mp4"
  op open "./clip.mp4" 1
  op frame 8
  op rect edit=1 "#FFFF00" "solid" "transparent" "" [10px 10px 90px 60px] 2 4 1
  op save ""
  op frame 90
  op filter edit=1 "sepia" ""
  op save ""
block 3 dir "./shots/"
  op open "./shots/" 3
  op filter edit=1 "bw" ""
  op rect edit=2 "#FFFF00" "solid" "transparent" "" [23px 14px 56px 90px] 2 4 1
  op save ""
block 4 glob "./shots/a*.png"
  op open "./shots/a*.png" 4
  op filter edit=1 "sepia" ""
  op save "./out/"

=== tour-style
--- script
# Templates: name a group of directives once, use it anywhere.

@stencil lines and rect:
    @use line #cccccc dashed, fill aqua, point red 2px
    @line (23, 14) (14px, 50%)
    @rect (10, 10) (90, 60)

# A quoted name may contain anything; an unquoted one is every word
# before the ':'.
@stencil "line-style":
    @use line #cccccc dashed, cm, fill aqua, point red 2px

# @1 and @2 are positional parameters; the count sets the arity.
@stencil line-style param:
    @use line @1 dashed, @2, fill aqua, point red 2px

@SOURCE ./somepath/someimage.png:
    @RECT (23, 14) (56, 90)
    @use stencil lines and rect:
    @use stencil "line-style":
    @use stencil line-style param red cm:
    @save
--- dump
block 0 file "./somepath/someimage.png"
  op open "./somepath/someimage.png" 1
  op rect edit=1 "#FFFF00" "solid" "transparent" "" [23px 14px 56px 90px] 2 4 1
  op line edit=2 "#cccccc" "dashed" "aqua" "red" [23px 14px 14px 50%] 2 2 0
  op rect edit=3 "#cccccc" "dashed" "aqua" "red" [10px 10px 90px 60px] 2 2 1
  op save ""
